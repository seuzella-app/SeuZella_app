    68	/**
    69	 * Decide qual provider usar:
    70	 * - Marcas manuais (intelbras, yale, papaiz, philco, samsung) → sempre manual
    71	 * - Marcas com API mas sem credenciais configuradas → fallback manual
    72	 * - Marcas com API + credenciais configuradas → adapter real (TTLock, Tuya, etc.)
    73	 *
    74	 * Retorna wrapper unificado para o orchestrator tratar igual.
    75	 */
    76	async function loadProvider(brand: LockBrand): Promise<UnifiedProviderModule | null> {
    77	  try {
    78	    const info = getBrandInfo(brand);
    79	    if (!info) return null;
    80	
    81	    // Marcas sem API → sempre manual
    82	    if (!info.apiAvailable) {
    83	      return wrapManualProvider(brand);
    84	    }
    85	
    86	    // Marcas com API mas sem credenciais no .env → cai no manual com warning
    87	    if (!hasCredentialsConfigured(brand)) {
    88	      console.info(`[locks] ${brand}: API disponível mas credenciais não configuradas — usando modo manual`);
    89	      return wrapManualProvider(brand);
    90	    }
    91	
    92	    // Marcas com API + credenciais → adapter real
    93	    switch (brand) {
    94	      case 'ttlock':
    95	        return wrapApiProvider(brand, ttlockProvider);
    96	      case 'tuya':
    97	        return wrapApiProvider(brand, tuyaProvider);
    98	      case 'igloohome':
    99	        return wrapApiProvider(brand, igloohomeProvider);
   100	      case 'nuki':
   101	        return wrapApiProvider(brand, nukiProvider);
   102	      case 'august':
   103	        return wrapApiProvider(brand, augustProvider);
   104	      default:
   105	        return wrapManualProvider(brand);
   106	    }
   107	  } catch (err) {
   108	    console.error(`[locks] Failed to load provider for ${brand}:`, err);
   109	    return wrapManualProvider(brand);
   110	  }
   111	}

   140	/**
   141	 * Wrapper para providers de API reais (TTLock, Tuya, Igloohome, Nuki, August).
   142	 * Encapsula o try/catch e o fallback gracioso para o modo manual em caso
   143	 * de falha de comunicação com o provedor.
   144	 *
   145	 * CRÍTICO: se a API do provedor falhar (timeout, 5xx, OAuth expirado),
   146	 * NÃO bloqueamos a geração de PIN — caímos no modo manual com warning,
   147	 * para que o host ainda possa gerar um PIN criptográfico e cadastrá-lo
   148	 * manualmente no app do fabricante.
   149	 */
   150	function wrapApiProvider(
   151	  brand: LockBrand,
   152	  providerModule:
   153	    | typeof ttlockProvider
   154	    | typeof tuyaProvider
   155	    | typeof igloohomeProvider
   156	    | typeof nukiProvider
   157	    | typeof augustProvider,
   158	): UnifiedProviderModule {
   159	  return {
   160	    async generatePin(input) {
   161	      // Se host forneceu PIN manual → usa direto (não chama API)
   162	      if (input.manualPin && input.manualPin.trim().length >= 4) {
   163	        return wrapManualProvider(brand).generatePin(input);
   164	      }
   165	
   166	      // Sem externalDeviceId → não pode chamar API (não sabemos qual lock)
   167	      if (!input.externalDeviceId) {
   168	        const warnings = [
   169	          `Dispositivo ${brand} sem externalDeviceId cadastrado — usando modo manual fallback.`,
   170	          `Cadastre o ID externo do dispositivo (lockId retornado pela API do provedor).`,
   171	        ];
   172	        const result = await manualProvider.generatePin({
   173	          deviceId: input.deviceId,
   174	          brand,
   175	          providerType: 'manual',
   176	          validFrom: input.validFrom,
   177	          validTo: input.validTo,
   178	          autoGenerate: true,
   179	        });
   180	        return {
   181	          ...result,
   182	          warnings: [...warnings, ...(result.warnings ?? [])],
   183	        };
   184	      }
   185	
   186	      // Tenta chamar a API do provedor
   187	      try {
   188	        const apiResult = await providerModule.generatePin({
   189	          externalDeviceId: input.externalDeviceId,
   190	          validFrom: input.validFrom,
   191	          validTo: input.validTo,
   192	          pin: input.manualPin,
   193	          guestName: input.guestName,
   194	        });
   195	
   196	        return {
   197	          pin: apiResult.pin,
   198	          source: 'api',
   199	          codeType: apiResult.codeType,
   200	          externalCodeId: apiResult.externalCodeId,
   201	        };
   202	      } catch (err) {
   203	        // Fallback gracioso — gera PIN criptográfico para host cadastrar manualmente
   204	        const errorMsg = (err as Error).message;
   205	        console.error(`[locks] ${brand} API generatePin failed:`, errorMsg);
   206	
   207	        const fallbackResult = await manualProvider.generatePin({
   208	          deviceId: input.deviceId,
   209	          brand,
   210	          providerType: 'manual',
   211	          validFrom: input.validFrom,
   212	          validTo: input.validTo,
   213	          autoGenerate: true,
   214	        });
   215	
   216	        return {
   217	          ...fallbackResult,
   218	          warnings: [
   219	            `⚠️ API ${brand} indisponível (${errorMsg.slice(0, 100)}).`,
   220	            `PIN gerado localmente — você precisa cadastrá-lo MANUALMENTE no app do provedor.`,
   221	            ...(fallbackResult.warnings ?? []),
   222	          ],
   223	        };
   224	      }
   225	    },
   226	
   227	    async revokePin(input) {
   228	      if (!input.externalDeviceId || !input.externalCodeId) return;
   229	      try {
   230	        await providerModule.revokePin({
   231	          externalDeviceId: input.externalDeviceId,
   232	          externalCodeId: input.externalCodeId,
   233	        });
   234	      } catch (err) {
   235	        // Revoke soft-fail — PIN já está revogado no DB local
   236	        console.warn(`[locks] ${brand} API revokePin soft-fail:`, (err as Error).message);
   237	      }
   238	    },
   239	  };
   240	}

   478	/**
   479	 * Gera um novo PIN para um dispositivo.
   480	 *
   481	 * Fluxo:
   482	 * 1. Carrega o dispositivo
   483	 * 2. Decide provider (API vs manual)
   484	 * 3. Se manual: gera PIN criptográfico OU usa PIN colado pelo host
   485	 * 4. Se API: chama o adapter específico
   486	 * 5. Persiste PIN no DB
   487	 * 6. Dispara entrega via WhatsApp (se guestPhone fornecido)
   488	 * 7. Registra auditoria
   489	 */
   490	export async function generatePin(input: GeneratePinInput): Promise<GeneratePinResult> {
   491	  const tenantId = await resolveTenantId();
   492	  if (!tenantId) throw new Error('Unauthorized');
   493	
   494	  // ── Rate limit: 50 PINs/hour per tenant ──────────────────────────────
   495	  // Prevents brute-force PIN enumeration and runaway scripts that would
   496	  // exhaust provider API quotas. Idempotent manual-PIN requests (same
   497	  // deviceId + same validFrom) are exempt via idempotency key.
   498	  const { pinRatelimit } = await import('@/lib/rate-limit');
   499	  const rateLimitKey = `pin:${tenantId}`;
   500	  const rateLimitResult = await pinRatelimit.limit(rateLimitKey);
   501	  if (!rateLimitResult.success) {
   502	    throw new Error('PIN_RATE_LIMIT_EXCEEDED');
   503	  }
   504	
   505	  const device = await getLockDevice(input.deviceId);
   506	  if (!device) throw new Error('Dispositivo não encontrado');
   507	
   508	  const dbAvailable = await isDatabaseAvailable();
   509	
   510	  // Carrega o provider apropriado
   511	  const provider = await loadProvider(device.brand);
   512	  if (!provider) {
   513	    throw new Error(`Não foi possível carregar o provider para a marca ${device.brand}`);
   514	  }
   515	
   516	  // Gera o PIN via provider
   517	  const providerResult = await provider.generatePin({
   518	    deviceId: device.id,
   519	    brand: device.brand,
   520	    providerType: device.providerType,
   521	    externalDeviceId: device.externalDeviceId,
   522	    oauthAccountId: device.oauthAccountId,
   523	    validFrom: input.validFrom,
   524	    validTo: input.validTo,
   525	    manualPin: input.manualPin,
   526	    autoGenerate: input.autoGenerate,
   527	    guestName: input.guestName,
   528	  });
   529	
   530	  const codeStr = providerResult.pin;
   531	  const {source} = providerResult;
   532	  const {codeType} = providerResult;
   533	  const {externalCodeId} = providerResult;
   534	
   535	  const status = derivePinStatus({
   536	    validFrom: input.validFrom,
   537	    validTo: input.validTo,
   538	  });
   539	
   540	  let codeId: string;
   541	  if (dbAvailable) {
   542	    const code = await db.lockCode.create({
   543	      data: {
   544	        deviceId: device.id,
   545	        tenantId,
   546	        guestName: input.guestName ?? null,
   547	        guestPhone: input.guestPhone ?? null,
   548	        bookingId: input.bookingId ?? null,
   549	        code: codeStr,
   550	        codeType,
   551	        source,
   552	        validFrom: input.validFrom,
   553	        validTo: input.validTo,
   554	        status,
   555	        note: input.note ?? null,
   556	        externalCodeId: externalCodeId ?? null,
   557	      },
   558	    });
   559	    codeId = code.id;
   560	
   561	    await db.lockEvent.create({
   562	      data: {
   563	        deviceId: device.id,
   564	        codeId,
   565	        tenantId,
   566	        eventType: 'generated',

   695	/** Revoga um PIN específico. */
   696	export async function revokePin(pinId: string, reason: string): Promise<boolean> {
   697	  const tenantId = await resolveTenantId();
   698	  if (!tenantId) throw new Error('Unauthorized');
   699	
   700	  const dbAvailable = await isDatabaseAvailable();
   701	  if (!dbAvailable) return true;
   702	
   703	  const pin = await db.lockCode.findFirst({
   704	    where: { id: pinId, tenantId },
   705	  });
   706	  if (!pin) return false;
   707	
   708	  // Se o PIN foi gerado via API, tenta revogar no provedor também
   709	  if (pin.source === 'api' && pin.externalCodeId) {
   710	    const device = await db.lockDevice.findFirst({
   711	      where: { id: pin.deviceId },
   712	      select: { brand: true, externalDeviceId: true },
   713	    });
   714	    if (device) {
   715	      const provider = await loadProvider(device.brand as LockBrand);
   716	      if (provider?.revokePin) {
   717	        await provider.revokePin({
   718	          externalDeviceId: device.externalDeviceId,
   719	          externalCodeId: pin.externalCodeId,
   720	        });
   721	      }
   722	    }
   723	  }
   724	
   725	  await db.lockCode.update({
   726	    where: { id: pinId },
   727	    data: {
   728	      revokedAt: new Date(),
   729	      revokedReason: reason,
   730	      status: 'revoked',
   731	    },
   732	  });
   733	
   734	  await db.lockEvent.create({
   735	    data: {
   736	      deviceId: pin.deviceId,
   737	      codeId: pinId,
   738	      tenantId,
   739	      eventType: 'revoked',
   740	      message: `PIN revogado: ${reason}`,
   741	      metadata: JSON.stringify({ reason, source: pin.source, externalCodeId: pin.externalCodeId }),
   742	    },
   743	  });
   744	
   745	  return true;
   746	}

   748	/**
   749	 * PÂNICO: Revoga TODOS os PINs ativos de um dispositivo.
   750	 *
   751	 * Use caso: hóspede relata vazamento de PIN, host suspeita de invasão,
   752	 * ou qualquer emergência onde é mais seguro cortar tudo e reemitir.
   753	 *
   754	 * Em paralelo, chama revokePin do provider para cada PIN que foi gerado via API.
   755	 * As chamadas ao provider são best-effort (soft-fail) — o DB local é a fonte
   756	 * da verdade para o status do PIN.
   757	 */
   758	export async function panicRevokeAllPins(deviceId: string, reason: string = 'Pânico acionado pelo host'): Promise<{
   759	  revokedCount: number;
   760	  providerRevokesAttempted: number;
   761	  providerRevokesFailed: number;
   762	}> {
   763	  const tenantId = await resolveTenantId();
   764	  if (!tenantId) throw new Error('Unauthorized');
   765	
   766	  const dbAvailable = await isDatabaseAvailable();
   767	  if (!dbAvailable) return { revokedCount: 0, providerRevokesAttempted: 0, providerRevokesFailed: 0 };
   768	
   769	  // 1. Busca todos os PINs ativos com externalCodeId (gerados via API)
   770	  const activeApiPins = await db.lockCode.findMany({
   771	    where: {
   772	      deviceId,
   773	      tenantId,
   774	      revokedAt: null,
   775	      source: 'api',
   776	      externalCodeId: { not: null },
   777	    },
   778	    select: { id: true, externalCodeId: true },
   779	  });
   780	
   781	  // 2. Busca o dispositivo para saber a brand + externalDeviceId
   782	  const device = await db.lockDevice.findFirst({
   783	    where: { id: deviceId, tenantId },
   784	    select: { brand: true, externalDeviceId: true },
   785	  });
   786	
   787	  // 3. Revoga todos no DB em paralelo
   788	  const result = await db.lockCode.updateMany({
   789	    where: {
   790	      deviceId,
   791	      tenantId,
   792	      revokedAt: null,
   793	    },
   794	    data: {
   795	      revokedAt: new Date(),
   796	      revokedReason: reason,
   797	      status: 'revoked',
   798	    },
   799	  });
   800	
   801	  // 4. Em paralelo, chama revokePin do provider para cada PIN de API
   802	  let providerRevokesFailed = 0;
   803	  if (device && activeApiPins.length > 0) {
   804	    const provider = await loadProvider(device.brand as LockBrand);
   805	    if (provider?.revokePin) {
   806	      const revokePromises = activeApiPins.map((pin) =>
   807	        provider.revokePin!({
   808	          externalDeviceId: device.externalDeviceId,
   809	          externalCodeId: pin.externalCodeId!,
   810	        }).catch((err) => {
   811	          console.warn(`[locks] panic revoke provider soft-fail for ${pin.id}:`, (err as Error).message);
   812	          providerRevokesFailed++;
   813	        }),
   814	      );
   815	      await Promise.allSettled(revokePromises);
   816	    }
   817	  }
   818	
   819	  await db.lockEvent.create({
   820	    data: {
   821	      deviceId,
   822	      tenantId,
   823	      eventType: 'panic_revoke',
   824	      message: `PÂNICO: ${result.count} PIN(s) revogado(s) — ${reason}`,
   825	      metadata: JSON.stringify({
   826	        reason,
   827	        count: result.count,
   828	        providerRevokesAttempted: activeApiPins.length,
   829	        providerRevokesFailed,
   830	      }),
   831	    },
   832	  });
   833	
   834	  return {
   835	    revokedCount: result.count,
   836	    providerRevokesAttempted: activeApiPins.length,
   837	    providerRevokesFailed,
   838	  };
   839	}

   963	    const providerModule = getProviderModule(device.brand as any);

   978	          message: 'Destravamento remoto via DDC',
   979	        },

   981	    } catch {}
   982	
   983	    return { success: true };

     2	// ZEHLA — Cron Secret Verification (simple shared-secret)
     3	// ============================================================
     4	//
     5	// Helper SIMPLIFICADO para validar requisições cron via CRON_SECRET.
     6	// Não substitui o cron-auth.ts (M2M Ed25519) — é um fallback prático
     7	// para proteção básica quando chaves M2M não estão configuradas.
     8	//
     9	// Padrões suportados:
    10	//   1. Header Authorization: Bearer <CRON_SECRET>
    11	//   2. Header x-internal-token: <CRON_SECRET>
    12	//   3. Query ?secret=<CRON_SECRET>
    13	//
    14	// Em desenvolvimento (NODE_ENV !== 'production' E sem CRON_SECRET configurado):
    15	//   - Permite acesso (para teste manual via curl)
    16	//
    17	// Em produção SEM CRON_SECRET configurado:
    18	//   - Rejeita com 503 Service Unavailable (fail-closed)
    19	//
    20	// Uso:
    21	//   import { verifyCronSecret } from '@/lib/security/cron-secret';
    22	//
    23	//   export async function GET(req: NextRequest) {
    24	//     const auth = verifyCronSecret(req);
    25	//     if (!auth.ok) return auth.response;
    26	//     // ... lógica do cron
    27	//   }
    28	// ============================================================
    29	
    30	import { NextRequest, NextResponse } from 'next/server';
    31	import crypto from 'crypto';
    32	
    33	export interface CronAuthResult {
    34	  ok: boolean;
    35	  response?: NextResponse;
    36	  source?: 'bearer' | 'header' | 'query' | 'dev-bypass';
    37	}
    38	
    39	/**
    40	 * Valida o secret do cron job.
    41	 * Em dev sem CRON_SECRET: permite (para teste manual).
    42	 * Em produção sem CRON_SECRET: rejeita com 503.
    43	 * Em produção com CRON_SECRET: valida via timing-safe comparison.
    44	 */
    45	export function verifyCronSecret(req: NextRequest): CronAuthResult {
    46	  const cronSecret = process.env.CRON_SECRET;
    47	  const isProduction = process.env.NODE_ENV === 'production';
    48	
    49	  // Dev bypass: sem CRON_SECRET configurado, permite em desenvolvimento
    50	  if (!cronSecret && !isProduction) {
    51	    return { ok: true, source: 'dev-bypass' };
    52	  }
    53	
    54	  // Produção sem secret configurado → fail-closed
    55	  if (!cronSecret && isProduction) {
    56	    console.error('[cron-secret] CRÍTICO: CRON_SECRET não configurado em produção. Rejeitando requisição.');
    57	    return {
    58	      ok: false,
    59	      response: NextResponse.json(
    60	        {
    61	          error: 'service_unavailable',
    62	          message: 'CRON_SECRET não configurado. Configure a env var no Vercel.',
    63	        },
    64	        { status: 503 }
    65	      ),
    66	    };
    67	  }
    68	
    69	  // Tenta header Authorization: Bearer <secret>
    70	  const authHeader = req.headers.get('authorization');
    71	  if (authHeader?.startsWith('Bearer ')) {
    72	    const token = authHeader.slice(7);
    73	    if (timingSafeEqual(token, cronSecret!)) {
    74	      return { ok: true, source: 'bearer' };
    75	    }
    76	  }
    77	
    78	  // Tenta header x-internal-token
    79	  const internalToken = req.headers.get('x-internal-token');
    80	  if (internalToken && timingSafeEqual(internalToken, cronSecret!)) {
    81	    return { ok: true, source: 'header' };
    82	  }
    83	
    84	  // Tenta query ?secret= — com encadeamento opcional para compatibilidade com Request padrão
    85	  let querySecret: string | null = null;
    86	  try {
    87	    // NextRequest tem nextUrl; Request padrão não tem
    88	    if (req.nextUrl?.searchParams) {
    89	      querySecret = req.nextUrl.searchParams.get('secret');
    90	    } else if (req.url) {
    91	      const url = new URL(req.url);
    92	      querySecret = url.searchParams.get('secret');
    93	    }
    94	  } catch {
    95	    // Se não conseguir parsear URL, ignora silenciosamente
    96	  }
    97	  if (querySecret && timingSafeEqual(querySecret, cronSecret!)) {
    98	    return { ok: true, source: 'query' };
    99	  }
   100	
   101	  // Rejeita silenciosamente (não revela motivo) — payload padronizado
   102	  return {
   103	    ok: false,
   104	    response: NextResponse.json(
   105	      { error: 'UNAUTHORIZED', code: 'UNAUTHORIZED' },
   106	      { status: 401, headers: { 'WWW-Authenticate': 'Bearer' } }
   107	    ),
   108	  };
   109	}
   110	
   111	function timingSafeEqual(a: string, b: string): boolean {
   112	  const bufA = Buffer.from(a);
   113	  const bufB = Buffer.from(b);
   114	  if (bufA.length !== bufB.length) return false;
   115	  return crypto.timingSafeEqual(bufA, bufB);
   116	}

   121	 */
   122	export function verifyVercelCron(req: NextRequest): CronAuthResult {
